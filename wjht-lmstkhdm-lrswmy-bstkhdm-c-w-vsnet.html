        // نقل
        private void transfert(ListBox l1, Button button1vers2, Button buttonEffacer1, ListBox l2, Button button2vers1, Button buttonEffacer2) {
            // نقل العناصر المحددة من القائمة l1 إلى القائمة l2
            for (int i = l1.SelectedIndices.Count - 1; i >= 0; i--) {
                // فهرس العنصر المحدد
                int index = l1.SelectedIndices[i];
                // إضافة إلى l2
                l2.Items.Add(l1.Items[index]);
                // حذف في l1
                l1.Items.RemoveAt(index);
            }
            // أزرار "مسح"
            buttonEffacer2.Enabled = l2.Items.Count != 0;
            buttonEffacer1.Enabled = l1.Items.Count != 0;
            // أزرار النقل
            button1vers2.Enabled = false;
}
    • السطر ب: تتلقى طريقة "transfert" ستة معلمات:
    • مرجع إلى القائمة التي تحتوي على العناصر المحددة والتي تسمى هنا l1. عند تشغيل التطبيق، تكون l1 إما listBox1 أو listBox2. نرى أمثلة على الاستدعاء، في السطرين 3 و8 من إجراءات النقل buttonXversY_Click.
    • مرجع إلى زر النقل المرتبط بالقائمة l1. على سبيل المثال، إذا كانت l1 هي listBox2، فسيكون button2vers1 (انظر الاستدعاء في السطر 8)
    • مرجع على زر مسح القائمة l1. على سبيل المثال، إذا كانت l1 هي listBox1، فسيكون المرجع هو buttonEffacer1 (انظر الاستدعاء في السطر 3)
    • المراجع الثلاثة الأخرى مماثلة ولكنها تشير إلى القائمة l2.
    • السطر د: تمثل المجموعة [ListBox].SelectedIndices مؤشرات العناصر المحددة في المكون [ListBox]. وهي مجموعة:
    • [ListBox].SelectedIndices.Count هو عدد العناصر في هذه المجموعة
    • [ListBox].SelectedIndices[i] هو العنصر رقم i في هذه المجموعة

    نقوم بتصفح المجموعة في الاتجاه المعاكس: نبدأ من نهاية المجموعة وننتهي عند بدايتها. وسنشرح السبب.

    • السطر f: مؤشر لعنصر محدد من القائمة l1
    • السطر h: تمت إضافة هذا العنصر إلى القائمة l2
    • السطر j: ويتم حذفه من القائمة l1. ولأنه تم حذفه، فإنه لم يعد محددًا. سيتم إعادة حساب المجموعة l1.SelectedIndices في السطر d. وستفقد العنصر الذي تم حذفه للتو. وستتغير أرقام جميع العناصر التي تلي هذا العنصر من n إلى n-1.
    • إذا كانت حلقة السطر (d) تصاعدية وكانت قد عالجت للتو العنصر رقم 0، فستعالج بعد ذلك العنصر رقم 1. لكن العنصر الذي كان يحمل الرقم 1 قبل حذف العنصر رقم 0، سيحمل بعد ذلك الرقم 0. وبالتالي سيتم تجاهله من قبل الحلقة.
    • إذا كانت حلقة السطر (d) تنازلية وكانت قد عالجت للتو العنصر رقم n، فستعالج بعد ذلك العنصر رقم n-1. بعد حذف العنصر رقم n، لا يتغير رقم العنصر رقم n-1. لذلك يتم معالجته في الدورة التالية من الحلقة.
    • السطور m-n: يعتمد حالة الأزرار [Effacer] على وجود أو عدم وجود عناصر في القوائم المرتبطة
    • السطر p: لم تعد القائمة l2 تحتوي على عناصر محددة: يتم إيقاف تشغيل زر النقل الخاص بها.

using System;
using System.Windows.Forms;
using Spring.Context;
using Spring.Context.Support;
using Metier;
using System.Text;

namespace Chap5 {
    static class Program {
        /// <summary>
        /// نقطة الدخول الرئيسية للتطبيق.
        /// </summary>
        [STAThread]
        static void Main() {
            // رمز تم إنشاؤه بواسطة Vs
            Application.EnableVisualStyles();
            Application.SetCompatibleTextRenderingDefault(false);

            // --------------- كود المطور
            // مثيلات الطبقات [metier] و [dao]
            IApplicationContext ctx = null;
            Exception ex = null;
            IImpotMetier metier = null;
            try {
                // سياق Spring
                ctx = ContextRegistry.GetContext();
                // يُطلب مرجع على الطبقة [metier]
                metier = (IImpotMetier)ctx.GetObject("metier");
            } catch (Exception e1) {
                // تخزين الاستثناء
                ex = e1;
            }
            // نموذج للعرض
            Form form = null;
            // هل حدث استثناء؟
            if (ex != null) {
                // نعم - يتم إنشاء رسالة الخطأ المراد عرضها
                StringBuilder msgErreur = new StringBuilder(String.Format("Chaîne des exceptions : {0}{1}", "".PadLeft(40, '-'), Environment.NewLine));
                Exception e = ex;
                while (e != null) {
                    msgErreur.Append(String.Format("{0}: {1}{2}", e.GetType().FullName, e.Message, Environment.NewLine));
                    msgErreur.Append(String.Format("{0}{1}", "".PadLeft(40, '-'), Environment.NewLine));
                    e = e.InnerException;
                }
                // إنشاء نافذة خطأ يتم تمرير رسالة الخطأ المراد عرضها إليها
                Form2 form2 = new Form2();
                form2.MsgErreur = msgErreur.ToString();
                // ستكون هذه هي النافذة التي سيتم عرضها
                form = form2;
            } else {
                // كل شيء سار على ما يرام
                // إنشاء واجهة رسومية [Form1] يتم تمرير المرجع إليها على الطبقة [metier]

                form1.Metier = metier;

            }

        }
